📦 安装
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  sudo dpkg -i ravbot_*.deb
  sudo apt-get install -f

🚀 初始化
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ravbot onboard              # 交互式配置向导
  ravbot gateway              # 前台启动
  ravbot health               # 验证运行

⚙️  配置 API 密钥
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  # OpenAI
  ravbot config set providers.openai.apiKey "sk-..."
  ravbot config set providers.openai.model "gpt-4"

  # Anthropic (Claude)
  ravbot config set providers.anthropic.apiKey "sk-ant-..."
  ravbot config set providers.anthropic.model "claude-3-5-sonnet-20241022"

  # Google (Gemini)
  ravbot config set providers.google.apiKey "AIza..."
  ravbot config set providers.google.model "gemini-2.0-flash-exp"

  # Qwen (通义千问)
  ravbot config set providers.qwen.apiKey "sk-..."
  ravbot config set providers.qwen.model "qwen-max"

💬 使用 Agent
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ravbot agent "你好"                    # 发送消息
  ravbot agent --model gpt-4 "问题"      # 指定模型
  ravbot agent --session my-session "问题" # 指定会话

📋 会话管理
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ravbot sessions list                   # 列出会话
  ravbot sessions show <id>              # 查看详情
  ravbot sessions delete <id>            # 删除会话
  ravbot sessions export <id> > file     # 导出会话

🔌 插件管理
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  # 创建插件
  mkdir -p ~/.ravbot/plugins/my-plugin
  cat > ~/.ravbot/plugins/my-plugin/ravbot.plugin.json << 'EOF'
  {
    "name": "my-plugin",
    "version": "1.0.0",
    "main": "index.js"
  }
  EOF

  # 启用插件
  ravbot config set plugins.allow '["my-plugin"]'
  ravbot gateway restart

  # 查看插件
  ravbot plugins list

📊 服务管理
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  # 安装为系统服务
  ravbot gateway install
  systemctl start ravbot
  systemctl enable ravbot
  systemctl status ravbot

  # 查看日志
  journalctl -u ravbot -f
  ravbot logs

🔧 常用配置
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ravbot config set gateway.port 19000
  ravbot config set gateway.controlUi.port 19001

  # 日志级别
  ravbot config set logging.level "debug"

  # 查看配置
  ravbot config get

  # 重新加载
  ravbot config reload

🎯 插件示例 (~/.ravbot/plugins/weather/index.js)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
export const tools = [{
  name: "get_weather",
  description: "查询城市天气",
  input_schema: {
    type: "object",
    properties: {
      city: { type: "string" }
    },
    required: ["city"]
  },
  execute: async (input) => {
    return `${input.city}天气: 晴天, 25°C`;
  }
}];

🚨 常见问题
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Q: 服务无法启动?
A: journalctl -u ravbot -n 50 --no-pager

Q: 端口冲突?
A: ravbot config set gateway.port 19000

Q: API 密钥无效?
A: ravbot config set providers.openai.apiKey "sk-..."

Q: Sidecar 无法启动?
A: node --version (需要 >= 18)
   ls -la /usr/share/ravbot/sidecar/dist/

Q: 插件加载失败?
A: ravbot plugins list
   cat ~/.ravbot/ravbot.json | grep -A 5 plugins
