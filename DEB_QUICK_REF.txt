└─────────────────────────────────────────────────────────────┘

📦 构建 DEB 包
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ./scripts/build-deb.sh              # 自动构建(需要 sudo)
  ./scripts/build-deb-local.sh        # 本地构建(无需 sudo)

  依赖: nodejs npm (用于编译 Sidecar)

🔧 安装
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  sudo dpkg -i dist/ravbot_*.deb
  sudo apt-get install -f             # 修复依赖(会安装 nodejs)

🚀 初始化(在当前用户下)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ravbot onboard                   # 初始化向导
  ravbot gateway                   # 前台运行

  # 或安装为系统服务
  ravbot gateway install
  systemctl start ravbot
  systemctl enable ravbot

🔍 验证
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ravbot --version
  ravbot health
  ravbot plugins list              # 查看插件状态
  ravbot dashboard

⚙️  配置(在当前用户下)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ravbot config set providers.openai.apiKey "sk-..."
  ravbot config set plugins.allow '["my-plugin"]'
  ravbot config reload

🔌 插件管理
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  # 安装插件
  mkdir -p ~/.ravbot/plugins/my-plugin

  # 启用插件
  ravbot config set plugins.allow '["my-plugin"]'

  # 查看插件
  ravbot plugins list
  curl http://localhost:18801/api/plugins/tools

📊 服务管理
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  systemctl start ravbot           # 启动
  systemctl stop ravbot            # 停止
  systemctl restart ravbot         # 重启
  journalctl -u ravbot -f          # 查看日志

🗑️  卸载
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  sudo apt-get remove ravbot       # 保留配置
  sudo apt-get purge ravbot        # 删除包
  rm -rf ~/.ravbot                 # 手动删除用户数据

📁 文件位置
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  /usr/bin/ravbot                  # 主程序 (C++)
  /usr/share/ravbot/sidecar/       # Node.js Sidecar
  /usr/share/ravbot/skills/        # 内置技能
  ~/.ravbot/                       # 配置和数据(当前用户)
  ~/.ravbot/plugins/               # 用户插件

🔌 Sidecar (插件运行时)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  自动包含在 DEB 包中
  支持 OpenClaw TypeScript 插件
  通过 TCP Socket 与 C++ 通信

  验证: ls -la /usr/share/ravbot/sidecar/dist/
